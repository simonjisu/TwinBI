# TPC-DS Query Examples

TPC-DS queries often use CTEs (Common Table Expressions), subqueries, and complex analytical patterns.

## Example 1: Customer Return Analysis with CTE (Similar to TPC-DS Q1)
WITH customer_total_return AS (
    SELECT 
        sr_customer_sk AS ctr_customer_sk,
        sr_store_sk AS ctr_store_sk,
        SUM(sr_return_amt) AS ctr_total_return
    FROM store_returns, date_dim
    WHERE sr_returned_date_sk = d_date_sk
        AND d_year = 2001
    GROUP BY sr_customer_sk, sr_store_sk
)
SELECT c_customer_id
FROM customer_total_return ctr1,
     store,
     customer
WHERE ctr1.ctr_total_return > (
        SELECT AVG(ctr_total_return) * 1.2
        FROM customer_total_return ctr2
        WHERE ctr1.ctr_store_sk = ctr2.ctr_store_sk
    )
    AND s_store_sk = ctr1.ctr_store_sk
    AND s_state = 'TN'
    AND ctr1.ctr_customer_sk = c_customer_sk
ORDER BY c_customer_id
LIMIT 100;

## Example 2: Top Items by Category with Window Functions
WITH item_sales AS (
    SELECT 
        i.i_item_sk,
        i.i_item_id,
        i.i_item_desc,
        i.i_category,
        SUM(ss.ss_ext_sales_price) AS total_sales,
        RANK() OVER (PARTITION BY i.i_category ORDER BY SUM(ss.ss_ext_sales_price) DESC) AS sales_rank
    FROM store_sales ss
    JOIN item i ON ss.ss_item_sk = i.i_item_sk
    JOIN date_dim d ON ss.ss_sold_date_sk = d.d_date_sk
    WHERE d.d_year = 2024
    GROUP BY i.i_item_sk, i.i_item_id, i.i_item_desc, i.i_category
)
SELECT 
    i_item_id,
    i_item_desc,
    i_category,
    total_sales
FROM item_sales
WHERE sales_rank <= 10
ORDER BY i_category, sales_rank;

## Example 3: Store Sales Analysis with Multiple CTEs
WITH yearly_sales AS (
    SELECT 
        s.s_store_sk,
        s.s_store_name,
        d.d_year,
        SUM(ss.ss_net_paid) AS annual_revenue,
        SUM(ss.ss_net_profit) AS annual_profit
    FROM store_sales ss
    JOIN store s ON ss.ss_store_sk = s.s_store_sk
    JOIN date_dim d ON ss.ss_sold_date_sk = d.d_date_sk
    GROUP BY s.s_store_sk, s.s_store_name, d.d_year
),
avg_store_sales AS (
    SELECT 
        d_year,
        AVG(annual_revenue) AS avg_annual_revenue
    FROM yearly_sales
    GROUP BY d_year
)
SELECT 
    ys.s_store_name,
    ys.d_year,
    ys.annual_revenue,
    ys.annual_profit,
    av.avg_annual_revenue,
    (ys.annual_revenue - av.avg_annual_revenue) AS revenue_vs_avg
FROM yearly_sales ys
JOIN avg_store_sales av ON ys.d_year = av.d_year
WHERE ys.annual_revenue > av.avg_annual_revenue * 1.1
ORDER BY ys.d_year, ys.annual_revenue DESC;

## Example 4: Customer Cohort Analysis with CASE Statements
SELECT 
    CASE 
        WHEN c.c_birth_year < 1960 THEN '< 1960'
        WHEN c.c_birth_year BETWEEN 1960 AND 1970 THEN '1960-1970'
        WHEN c.c_birth_year BETWEEN 1971 AND 1980 THEN '1971-1980'
        WHEN c.c_birth_year BETWEEN 1981 AND 1990 THEN '1981-1990'
        ELSE '> 1990'
    END AS age_cohort,
    ca.ca_state,
    COUNT(DISTINCT c.c_customer_sk) AS customer_count,
    SUM(ss.ss_ext_sales_price) AS total_sales,
    AVG(ss.ss_ext_sales_price) AS avg_transaction_value
FROM store_sales ss
JOIN customer c ON ss.ss_customer_sk = c.c_customer_sk
JOIN customer_address ca ON c.c_current_addr_sk = ca.ca_address_sk
JOIN date_dim d ON ss.ss_sold_date_sk = d.d_date_sk
WHERE d.d_year IN (2023, 2024)
GROUP BY age_cohort, ca.ca_state
HAVING COUNT(DISTINCT c.c_customer_sk) > 100
ORDER BY total_sales DESC;

## Example 5: Time-based Sales Analysis with Self-Join
WITH daily_sales AS (
    SELECT 
        d.d_date,
        d.d_year,
        d.d_moy,
        i.i_category,
        SUM(ss.ss_ext_sales_price) AS daily_total
    FROM store_sales ss
    JOIN date_dim d ON ss.ss_sold_date_sk = d.d_date_sk
    JOIN item i ON ss.ss_item_sk = i.i_item_sk
    GROUP BY d.d_date, d.d_year, d.d_moy, i.i_category
)
SELECT 
    ds1.i_category,
    ds1.d_year,
    ds1.d_moy AS month,
    SUM(ds1.daily_total) AS current_month_sales,
    LAG(SUM(ds1.daily_total), 1) OVER (PARTITION BY ds1.i_category ORDER BY ds1.d_year, ds1.d_moy) AS prev_month_sales,
    (SUM(ds1.daily_total) - LAG(SUM(ds1.daily_total), 1) OVER (PARTITION BY ds1.i_category ORDER BY ds1.d_year, ds1.d_moy)) AS sales_change
FROM daily_sales ds1
GROUP BY ds1.i_category, ds1.d_year, ds1.d_moy
ORDER BY ds1.i_category, ds1.d_year, ds1.d_moy;

## Example 6: Cross-dimensional Analysis with Subquery
SELECT 
    s.s_store_name,
    i.i_category,
    COUNT(DISTINCT ss.ss_customer_sk) AS unique_customers,
    SUM(ss.ss_quantity) AS total_quantity,
    SUM(ss.ss_ext_sales_price) AS total_sales
FROM store_sales ss
JOIN store s ON ss.ss_store_sk = s.s_store_sk
JOIN item i ON ss.ss_item_sk = i.i_item_sk
JOIN date_dim d ON ss.ss_sold_date_sk = d.d_date_sk
WHERE d.d_year = 2024
    AND ss.ss_ext_sales_price > (
        SELECT AVG(ss2.ss_ext_sales_price)
        FROM store_sales ss2
        WHERE ss2.ss_item_sk = ss.ss_item_sk
    )
GROUP BY s.s_store_name, i.i_category
ORDER BY total_sales DESC
LIMIT 50;
